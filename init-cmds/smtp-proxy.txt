# set sendgrid api key
head -n 1 | tr -d '\n' | podman secret create smtp-proxy-pass -

# create utils dir
mkdir -p /usr/local/smtp-proxy-utils
chcon -u system_u -t container_file_t /usr/local/smtp-proxy-utils

# link mail utils
ln -s /usr/local/smtp-proxy-utils/mail-in-ctr.sh /usr/bin/mail
ln -s /usr/local/smtp-proxy-utils/sendmail-in-ctr.sh /usr/bin/sendmail

# link mail logs service
systemctl link /usr/local/smtp-proxy-utils/email-unit-invocation-logs@.service

# configure email address
mkdir -p /etc/systemd/system/email-unit-invocation-logs@.service.d
echo -e '[Service]\nEnvironment=EMAIL_TGTS=user@example.com' > \
	/etc/systemd/system/email-unit-invocation-logs@.service.d/10-set-email.conf
